// Hotspots / ホットスポット
// Pack: lp2 (Landing page & web effects, part 2) · element 19 · Category: Showcase effects
// What happens: White dots pulse on the product picture; clicking one opens a tooltip about that part, and clicking another dot swaps the tooltip.
// Motion spec:
//   - A 511 × 168 viewport: a product picture (a backpack) on the left, name and price on the right, with two dots on the picture (white circle, radius 8, drop shadow; orange center, radius 4). One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - Pulse: from each dot a white ring (width 2) grows from radius 8 to 20 while fading from 0.75 to 0 opacity, every 1.57 s (a third of 4.7 s), two rings half a period apart.
//   - At 0.9 s the upper dot is clicked: a tooltip (136 × 48, radius 8, with a pointer toward the dot) appears to its left over 0.25 s (cubic ease-out), rising 6 px.
//   - At 2.4 s the lower dot is clicked: the first tooltip fades over 0.25 s and the next one opens to the right of the lower dot. A second click at 3.8 s closes it.
// When to use it: Use it on products, floor plans or maps, to explain different spots within one picture.
// Tags: hotspots, image, tooltip, pulse, product, interactive-image
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#163
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 19
// 19 ホットスポット（製品写真の上で点が脈打ち、押すと説明の吹き出しが出る）
CELLS[18] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const D1 = [186, 52], D2 = [150, 122];
  const o1 = t => { t = mod(t, CYC); return t >= 0.9 && t < 2.4; }, o2 = t => { t = mod(t, CYC); return t >= 2.4 && t < 3.8; };
  const q1 = trans(o1, u, 0.25), q2 = trans(o2, u, 0.25);
  const cur = t => path([[0, 280, 205], [0.6, D1[0], D1[1]], [1.15, D1[0], D1[1]], [2.05, D2[0], D2[1]], [3.95, D2[0], D2[1]], [4.45, 280, 205], [4.7, 280, 205]], t);
  const [cx, cy] = cur(u);
  view(k, () => {
    k.fillStyle = '#EEF1F6'; k.fillRect(0, 0, 300, VH);
    // リュック（平面の絵）
    k.strokeStyle = C.ink; k.lineWidth = 6; k.lineCap = 'round'; k.beginPath(); k.arc(150, 30, 16, Math.PI, 0); k.stroke();
    k.fillStyle = '#3B64E8'; rr(k, 92, 26, 116, 134, 28); k.fill();
    k.fillStyle = '#2E50C2'; rr(k, 108, 92, 84, 56, 14); k.fill();
    k.strokeStyle = '#FFD9C9'; k.lineWidth = 2.5; k.beginPath(); k.moveTo(116, 104); k.lineTo(184, 104); k.stroke();
    k.fillStyle = C.acc; rr(k, 140, 54, 20, 8, 4); k.fill();
    txt(k, 'トラベルリュック', 322, 58, `800 16px ${JP}`, C.ink);
    txt(k, '¥12,800', 322, 88, `800 19px ${EN}`, C.acc);
    bars(k, 322, 104, 160, 3);
    [D1, D2].forEach(([x, y], i) => {
      for (let r = 0; r < 2; r++) { const f = mod(u / (CYC / 3) + r * 0.5, 1); k.strokeStyle = `rgba(255,255,255,${0.75 * (1 - f)})`; k.lineWidth = 2; k.beginPath(); k.arc(x, y, 8 + 12 * f, 0, 7); k.stroke(); }
      k.save(); k.shadowColor = 'rgba(28,35,51,.3)'; k.shadowBlur = 6; k.shadowOffsetY = 2; k.fillStyle = '#fff'; k.beginPath(); k.arc(x, y, 8, 0, 7); k.fill(); k.restore();
      k.fillStyle = C.acc; k.beginPath(); k.arc(x, y, 4, 0, 7); k.fill();
    });
    const tip = (x, y, q, title, left) => {
      if (q <= 0) return;
      k.save(); k.globalAlpha = q; k.translate(x, y + 6 * (1 - q));
      const L = left ? 0 : 1;   // 吹き出しを点の左（left）か右に出す
      const bx = L ? 0 : -136;
      k.save(); k.shadowColor = 'rgba(28,35,51,.22)'; k.shadowBlur = 12; k.shadowOffsetY = 4; k.fillStyle = '#fff'; rr(k, bx, -24, 136, 48, 8); k.fill();
      k.beginPath(); if (L) { k.moveTo(0, -6); k.lineTo(-8, 0); k.lineTo(0, 6); } else { k.moveTo(0, -6); k.lineTo(8, 0); k.lineTo(0, 6); } k.fill(); k.restore();
      txt(k, title, bx + 12, -6, `800 13px ${JP}`, C.ink); bars(k, bx + 12, 6, 100, 1, C.faint, 5);
      k.restore();
    };
    tip(D1[0] - 16, D1[1], q1, '雨をはじく布', true);
    tip(D2[0] + 16, D2[1], q2, 'PCが入るポケット', false);
    click(k, cx, cy, u, [0.9, 2.4, 3.8]);
  });
};
