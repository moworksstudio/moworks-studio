// Cutaway / カットアウェイ
// Pack: camera2 (Camera moves & transitions, part 2) · element 21 · Category: Transitions & Editing
// What happens: From the speaker to the listening audience and back; what is cut in is a different subject.
// Motion spec:
//   - Join: two straight cuts (0 frames); to the other subject at 1.7 s and back to the scene at 3.0 s (1.3 s away)
//   - Main shot (A): fixed camera (height 1.4 m); a person 5.5 m away talks, swinging arms and legs by 0.45·sin(u×6) rad, with a grey lectern 0.9 m to the right
//   - Other subject (CUTAWAY): the listening audience, five people 3.1–4.6 m away in blue and pale blue with both arms up, bouncing by |sin(t×9+i)|×5 cm; no background (a different place)
//   - Bottom strip: A (grey), other subject (blue), A (grey) with a playhead (17 Insert shot shows a detail of the same scene)
//   - Top-right readout shows "SHOT A" → "CUTAWAY" → "SHOT A"; at 4.7 s it cuts straight back to the start
// When to use it: To show a reaction, or to hide a time skip or a removed slip.
// Tags: cutaway, reaction shot, b-roll, cut away, audience, edit
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#189
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 21
// 21 カットアウェイ: 話す人物から、別の対象（聞いている客席）へ切り替えて戻る。17 インサートは同じ場面の細部
const CUT = [1.7, 3.0];
CELLS[20] = shot((k, w, h, u) => {
  const away = u >= CUT[0] && u < CUT[1];
  if (!away) world(k, w, h, cam({ y: 1.4 }), { person: { x: 0, z: 5.5, walk: 0.45 * Math.sin(u * 6) }, extra: [[0.9, 0, 5.6, (kk) => { kk.fillStyle = C.g3; kk.fillRect(-0.3, 0, 0.6, 1.1); kk.fillRect(-0.4, 1.05, 0.8, 0.1); }]] });
  else {
    const t = u - CUT[0], ex = [];
    [[-1.6, 3.2], [-0.2, 3.6], [1.3, 3.1], [0.6, 4.4], [-1.1, 4.6]].forEach(([x, z], i) => ex.push([x, 0, z, (kk) => { kk.save(); kk.translate(0, Math.abs(Math.sin(t * 9 + i)) * 0.05); person(kk, 1, { armUp: true, col: i % 2 ? C.blue : C.blueM }); kk.restore(); }]));
    world(k, w, h, cam({ y: 1.6, pitch: -0.08 }), { buildings: false, trees: false, mountains: false, sun: null, extra: ex });
  }
  strip(k, w, h, [[0, CUT[0], 'A', C.g2], [CUT[0], CUT[1], '別の対象', C.blue], [CUT[1], CYC, 'A', C.g2]], u);
  return away ? 'CUTAWAY' : 'SHOT A';
});
