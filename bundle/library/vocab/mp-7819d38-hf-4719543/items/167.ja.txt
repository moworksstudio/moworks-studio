// スクロールダウンの合図 / Scroll-down cue
// パック: lp2 (LP・Webサイトの演出 第2弾) · element 23 · カテゴリ: 見せ場の演出
// 何が起きるか: ファーストビューの下で、「SCROLL」の文字とマウスの形の中の点、両わきの矢印が、下へ促す動きをくり返す。
// 動きの仕様:
//   - 表示域 511×168。ファーストビューの下の中央に「SCROLL」（11px・字間3px）とマウスの形の枠（24×38・角丸12・線幅2.2）。1回の動きは1.57秒（4.7秒に3回、パックの周期9.4秒で6回）
//   - 枠の中の朱の点（5×9）: 1回の最初の70%で14px下がり（easeInOut 2乗）、45%〜75%の間に消えて、次の回の頭に上で現れる
//   - 枠の左右40pxに下向きの矢印（線幅2.5）を2つずつ、半周期ずらして24px下へ流す。不透明度は sin で 0→0.8→0（出て、消える）
// 使いどころ: ファーストビューが画面いっぱいで、その下にも続きがあることを伝えたいとき。
// タグ: scroll-cue, scroll-indicator, scroll-down, hero, arrow, loop
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#167
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 23
// 23 スクロールダウンの合図（ファーストビューの下で「SCROLL」とマウスの形の中の点が、下へ促す動きをくり返す）
CELLS[22] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const T = CYC / 3, f = mod(u, T) / T;
  view(k, () => {
    k.fillStyle = '#FFF4EA'; k.fillRect(0, 0, VW, VH);
    txt(k, 'ようこそ、', 40, 40, `800 22px ${JP}`, C.ink); txt(k, '山の上の宿へ。', 40, 70, `800 22px ${JP}`, C.ink);
    bars(k, 40, 84, 170, 2, '#EAD9CB');
    pic(k, 286, 10, 200, 92, 1, 10);
    // 合図
    const mx = VW / 2;
    txt(k, 'SCROLL', mx, 116, `800 11px ${EN}`, C.ink, 'center', 'middle', '3px');
    k.strokeStyle = C.ink; k.lineWidth = 2.2; rr(k, mx - 12, 126, 24, 38, 12); k.stroke();
    const dy = eIOs(clamp(f / 0.7)) * 14, da = 1 - clamp((f - 0.45) / 0.3);
    k.fillStyle = `rgba(255,90,44,${da})`; rr(k, mx - 2.5, 133 + dy, 5, 9, 2.5); k.fill();
    // 両わきの矢印も下へ流れる
    [mx - 40, mx + 40].forEach(x => { for (let j = 0; j < 2; j++) { const ff = mod(f + j * 0.5, 1), yy = 128 + ff * 24, al = Math.sin(Math.PI * ff); k.strokeStyle = `rgba(28,35,51,${0.8 * al})`; k.lineWidth = 2.5; k.lineCap = 'round'; k.lineJoin = 'round'; k.beginPath(); k.moveTo(x - 8, yy); k.lineTo(x, yy + 8); k.lineTo(x + 8, yy); k.stroke(); } });
  });
};
