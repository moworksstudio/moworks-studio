// バレットタイム / Bullet time
// パック: camera2 (カメラワークと切り替え 第2弾) · element 23 · カテゴリ: 切り替え・編集
// 何が起きるか: 跳躍の頂点で時間がほぼ止まり、その間にカメラが人物の周りを回り込む。
// 動きの仕様:
//   - 映像時間: 0〜1.5秒は等速、1.5〜3.3秒は0.1倍（1.8秒で0.18秒ぶんだけ進む）、3.3秒以降は等速に戻る（映像時間 = u - 1.62）
//   - 被写体: 人物（9m先）は映像時間0.6〜2.6秒に放物線で跳ぶ（頂点1.1m、両腕を上げる）。スローの間は頂点付近（進み45〜54%）で止まったように見える。葉9枚も同じ映像時間で落ちる
//   - カメラ: スローの間に人物を中心とする半径6mの円周を -0.6rad → +0.6rad（約69°）回り込む。3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当）。常に中心を向き、高さ1.6m、ピッチ -0.05rad
//   - 次の4.7秒は逆向き（+0.6 → -0.6rad）に回るので、9.4秒で継ぎ目なく戻る（#030 オービットは時間が止まらない）
//   - 足元に半径1.3mの点線の円、周囲に半径10mの木の輪。右上に「×1.0」→「BULLET ×0.1」→「×1.0」
// 使いどころ: 決めの瞬間を立体的に見せるとき、アクションの見せ場。
// タグ: bullet time, time slice, frozen moment, orbit, slow motion, matrix, effect
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#191
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 23
// 23 バレットタイム: 跳躍の頂点で時間がほぼ止まり、その間にカメラが人物の周りを回り込む（#030 オービットは時間が止まらない）
CELLS[22] = shot((k, w, h, u, cyc) => {
  const T0 = 1.5, T1 = 3.3, SLOW = 0.1;
  const tj = u < T0 ? u : u < T1 ? T0 + (u - T0) * SLOW : u - (T1 - T0) * (1 - SLOW);   // 実時間→映像時間
  const q = P(tj, 0.6, 2.6), e = eIO(P(u, T0, T1));
  const a = cyc % 2 === 0 ? lerp(-0.6, 0.6, e) : lerp(0.6, -0.6, e), R = 6, P0 = [0, 0, 9];
  const c = cam({ x: P0[0] + Math.sin(a) * R, z: P0[2] - Math.cos(a) * R, yaw: -a, y: 1.6, pitch: -0.05 });
  world(k, w, h, c, {
    person: { z: 9, y: jumpY(q), armUp: q > 0 && q < 1, face: a }, ring: [0, 9, 10], trees: false, extra: leaves(tj, 1.5).map(l => [l[0], l[1], l[2] + 3, l[3]]),
    after: (kk, cc) => { kk.strokeStyle = C.blueM; kk.lineWidth = 2; kk.setLineDash([5, 5]); kk.beginPath(); for (let i = 0; i <= 48; i++) { const b = i / 48 * Math.PI * 2; const qq = proj(cc, Math.sin(b) * 1.3, 0, 9 + Math.cos(b) * 1.3); i ? kk.lineTo(qq.x, qq.y) : kk.moveTo(qq.x, qq.y); } kk.stroke(); kk.setLineDash([]); }
  });
  return u >= T0 && u < T1 ? `BULLET ×${SLOW.toFixed(1)}` : '×1.0';
});
