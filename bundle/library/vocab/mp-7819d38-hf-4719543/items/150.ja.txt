// 何が起きるか: 本文をスクロールすると、左の目次のうち、いま読んでいる見出しの項目に色が付き、次の見出しに入ると印が移る。
// 動きの仕様:
//   - 表示域 511×168。左に目次（幅150px・4項目）、右に本文。見出し4つはページ上で100pxおき。スクロール量: 0.3〜0.9秒 0→100、1.35〜1.95秒 →200、2.4〜3.0秒 →300、3.5〜4.4秒 300→0（easeInOut 3乗）。4.7秒でひと回り（パックの周期9.4秒で2回）
//   - 今の項目＝上端が「スクロール量＋40px」より上にある見出しのうち最後の1つ（判定線は画面の上から40px）
//   - 今の項目が変わると0.25秒（easeOut 3乗）で、目次の項目に薄い朱の地（126×26・角丸6）と左端の朱の線（幅4）を付け、文字を灰色→朱・太字にする。前の項目は同じ0.25秒で戻る
//   - 目次は動かず、本文だけがスクロールする。右端のスクロールバーも連動
// 使いどころ: 長い記事・ドキュメント・FAQで、いまどこを読んでいるかを全体の中で示したいとき。
// タグ: scrollspy, table-of-contents, toc, navigation, active-state, scroll
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#150
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 06
// 06 スクロールスパイ（左の目次のうち、いま読んでいる見出しの項目が光る）
CELLS[5] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const y = steps([[0.3, 0], [0.9, 100], [1.35, 100], [1.95, 200], [2.4, 200], [3.0, 300], [3.5, 300], [4.4, 0]], u);
  const NAMES = ['はじめに', '使い方', '料金', 'よくある質問'], TX = 160;
  const act = t => { const yy = steps([[0.3, 0], [0.9, 100], [1.35, 100], [1.95, 200], [2.4, 200], [3.0, 300], [3.5, 300], [4.4, 0]], t); let a = 0; for (let i = 0; i < 4; i++) if (i * 100 <= yy + 40) a = i; return a; };
  view(k, () => {
    k.save(); k.beginPath(); k.rect(TX - 10, 0, VW, VH); k.clip(); k.translate(0, -y);
    NAMES.forEach((s, i) => {
      const y0 = i * 100 + 14;
      txt(k, `${i + 1}. ${s}`, TX + 14, y0 + 22, `800 16px ${JP}`, C.ink);
      bars(k, TX + 14, y0 + 38, 300, 3, C.faint, 6, 12, 0.6);
    });
    k.restore();
    k.fillStyle = C.soft; k.fillRect(0, 0, TX - 10, VH);
    txt(k, '目次', 20, 30, `800 12px ${JP}`, C.sub);
    const a = act(u);
    NAMES.forEach((s, i) => {
      const q = trans(t => act(t) === i, u, 0.25), iy = 54 + i * 30;
      if (q > 0) { k.fillStyle = `rgba(255,229,219,${q})`; rr(k, 12, iy - 13, TX - 34, 26, 6); k.fill(); k.fillStyle = C.acc; k.globalAlpha = q; rr(k, 12, iy - 13, 4, 26, 2); k.fill(); k.globalAlpha = 1; }
      txt(k, s, 26, iy + 1, `${a === i ? 800 : 600} 13px ${JP}`, mix('#7D879B', C.acc, q), 'left', 'middle');
    });
    sbar(k, y / 300, 0.36);
  });
};
