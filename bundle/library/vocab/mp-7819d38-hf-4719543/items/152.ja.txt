// スクロールで拡大 / Zoom on scroll
// パック: lp2 (LP・Webサイトの演出 第2弾) · element 08 · カテゴリ: スクロールで起きる演出
// 何が起きるか: スクロールに合わせて、ページ中央の小さな写真が窓いっぱいまで広がり、周りの文字が消えて、写真の上に見出しが出る。
// 動きの仕様:
//   - 表示域 511×168。スクロールの進み p: 0.3〜2.3秒で 0→1、3.0〜4.4秒で 1→0（easeInOut 3乗）。4.7秒でひと回り（パックの周期9.4秒で2回）
//   - 写真の枠: 中央の 155×90（位置 x=178, y=58・角丸8）から窓いっぱいの 511×168（角丸0）まで、位置・幅・高さ・角丸を p に比例して変える
//   - 周りの見出しと文字は p が0〜0.45の間に消える（不透明度 1 − 2.2p）
//   - p が0.7〜1の間に、写真の中央に白い見出し（26px・字間2px）が現れる。右端のスクロールバーも連動
// 使いどころ: 旅・不動産・ブランドサイトなどで、写真の世界に入り込む感じを出したいとき。
// タグ: zoom, scroll, image, scale, scroll-driven, hero, fullscreen
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#152
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 08
// 08 スクロールで拡大（スクロールに合わせて、小さな写真が窓いっぱいまで広がる）
CELLS[7] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.3, 1], [3.0, 1], [4.4, 0]], u);
  const x = lerp(178, 0, p), y = lerp(58, 0, p), ww = lerp(155, VW, p), hh = lerp(90, VH, p);
  view(k, () => {
    const a = 1 - clamp(p * 2.2);
    k.save(); k.globalAlpha = a;
    txt(k, '景色の中へ', VW / 2, 38, `800 19px ${JP}`, C.ink, 'center');
    bars(k, 34, 66, 120, 4, C.faint, 6, 14); bars(k, 357, 66, 120, 4, C.faint, 6, 14);
    k.restore();
    pic(k, x, y, ww, hh, 0, lerp(8, 0, p));
    const b = clamp((p - 0.7) / 0.3);
    if (b > 0) { k.save(); k.globalAlpha = b; txt(k, '絶景へようこそ', VW / 2, VH / 2 + 2, `800 26px ${JP}`, '#fff', 'center', 'middle', '2px'); k.restore(); }
    sbar(k, p * 0.6, 0.4);
  });
};
