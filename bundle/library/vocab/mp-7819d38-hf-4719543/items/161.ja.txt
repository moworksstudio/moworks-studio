// ビュートランジション / View transition (shared element)
// パック: lp2 (LP・Webサイトの演出 第2弾) · element 17 · カテゴリ: 見せ場の演出
// 何が起きるか: 一覧の写真を押すと、その写真が1枚のまま大きくなって詳細ページの上部の写真になり、周りが詳細ページに入れ替わる。戻ると元の位置へ縮む。
// 動きの仕様:
//   - 表示域 511×168。一覧に写真3枚（143×80・角丸6）。0.75秒に真ん中の写真をクリック、3.05秒に詳細の「← 一覧へ」をクリック。4.7秒でひと回り（パックの周期9.4秒で2回）
//   - 行き: 0.85〜1.45秒（0.6秒・easeInOut 3乗）で、押した写真の位置と大きさを一覧の枠から詳細ページ上部の 479×94（角丸8）まで連続して変える。写真は1枚のまま移る
//   - ほかの写真と見出しは進み0〜62%で消え、詳細の見出し・本文・戻るボタンは進み60〜100%で現れる。進み50%でURLを /works から /works/02 に変える
//   - 戻り: 3.15〜3.75秒（0.6秒）で同じ動きを逆にたどり、一覧の位置へ戻す
//   - 一覧と詳細の写真に同じ view-transition-name を付けたときの View Transitions API の見え方
// 使いどころ: 一覧と詳細を行き来するギャラリーや商品ページで、どの項目を開いたかを見失わせたくないとき。
// タグ: view-transition, shared-element, page-transition, morph, gallery, navigation
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#161
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 17
// 17 ビュートランジション（一覧の写真を押すと、その写真がそのまま大きくなって詳細ページの上部になる）
CELLS[16] = (k, w, h, u) => {
  u = snapU(u);
  const e = u < 2.0 ? eIO(P(u, 0.85, 1.45)) : 1 - eIO(P(u, 3.15, 3.75));
  frame(k, e > 0.5 ? 'example.com/works/02' : 'example.com/works');
  const G = [24, 184, 344], GY = 44, GW = 143, GH = 80, NM = ['雪の山小屋', '海辺のカフェ', '夜の展望台'], PV = [0, 1, 2];
  const T = { x: 16, y: 10, w: VW - 32, h: 94 };
  const cur = t => path([[0, 420, 205], [0.5, 262, 88], [0.95, 262, 88], [1.6, 380, 150], [2.55, 380, 150], [2.9, 62, 30], [3.2, 62, 30], [3.9, 420, 205], [4.7, 420, 205]], t);
  const [cx, cy] = cur(u);
  view(k, () => {
    const ga = 1 - clamp(e * 1.6);
    k.save(); k.globalAlpha = ga;
    txt(k, 'WORKS', 24, 30, `800 15px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    for (let i = 0; i < 3; i++) { if (i !== 1) pic(k, G[i], GY, GW, GH, PV[i], 6); txt(k, NM[i], G[i], GY + GH + 20, `800 13px ${JP}`, C.ink); bars(k, G[i], GY + GH + 30, 90, 1, C.faint, 5); }
    k.restore();
    const x = lerp(G[1], T.x, e), y = lerp(GY, T.y, e), ww = lerp(GW, T.w, e), hh = lerp(GH, T.h, e);
    pic(k, x, y, ww, hh, 1, lerp(6, 8, e));
    const da = clamp((e - 0.6) / 0.4);
    if (da > 0) {
      k.save(); k.globalAlpha = da;
      k.fillStyle = 'rgba(255,255,255,.92)'; rr(k, 26, 19, 74, 22, 11); k.fill(); txt(k, '← 一覧へ', 63, 30.5, `700 11px ${JP}`, C.ink, 'center', 'middle');
      txt(k, '海辺のカフェ', 24, 134, `800 19px ${JP}`, C.ink); bars(k, 200, 122, 280, 3, C.faint, 6, 12, 0.7);
      k.restore();
    }
    click(k, cx, cy, u, [0.75, 3.05]);
  });
};
