// Bullet time / バレットタイム
// Pack: camera2 (Camera moves & transitions, part 2) · element 23 · Category: Transitions & Editing
// What happens: Time nearly stops at the top of a jump while the camera circles around the person.
// Motion spec:
//   - Footage time: runs normally for 0–1.5 s, at 0.1× from 1.5 s to 3.3 s (only 0.18 s of footage in 1.8 s), then normally again (footage time = u - 1.62)
//   - Subject: a person 9 m away jumps on a parabola over footage time 0.6–2.6 s (peak 1.1 m, arms raised); during the slow part they seem to hang near the top (45–54% through); nine leaves fall on the same footage time
//   - Camera: during the slow part it travels on a circle of radius 6 m around the person from -0.6 rad to +0.6 rad (about 69°), cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)); always facing the centre, height 1.6 m, pitch -0.05 rad
//   - The next 4.7 s pass circles the other way (+0.6 → -0.6 rad), so the 9.4 s loop is seamless (#030 Orbit does not stop time)
//   - A dashed circle of radius 1.3 m on the ground and a ring of trees at 10 m; top-right readout shows "×1.0" → "BULLET ×0.1" → "×1.0"
// When to use it: To show a decisive moment in three dimensions, or as the highlight of an action scene.
// Tags: bullet time, time slice, frozen moment, orbit, slow motion, matrix, effect
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#191
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 23
// 23 バレットタイム: 跳躍の頂点で時間がほぼ止まり、その間にカメラが人物の周りを回り込む（#030 オービットは時間が止まらない）
CELLS[22] = shot((k, w, h, u, cyc) => {
  const T0 = 1.5, T1 = 3.3, SLOW = 0.1;
  const tj = u < T0 ? u : u < T1 ? T0 + (u - T0) * SLOW : u - (T1 - T0) * (1 - SLOW);   // 実時間→映像時間
  const q = P(tj, 0.6, 2.6), e = eIO(P(u, T0, T1));
  const a = cyc % 2 === 0 ? lerp(-0.6, 0.6, e) : lerp(0.6, -0.6, e), R = 6, P0 = [0, 0, 9];
  const c = cam({ x: P0[0] + Math.sin(a) * R, z: P0[2] - Math.cos(a) * R, yaw: -a, y: 1.6, pitch: -0.05 });
  world(k, w, h, c, {
    person: { z: 9, y: jumpY(q), armUp: q > 0 && q < 1, face: a }, ring: [0, 9, 10], trees: false, extra: leaves(tj, 1.5).map(l => [l[0], l[1], l[2] + 3, l[3]]),
    after: (kk, cc) => { kk.strokeStyle = C.blueM; kk.lineWidth = 2; kk.setLineDash([5, 5]); kk.beginPath(); for (let i = 0; i <= 48; i++) { const b = i / 48 * Math.PI * 2; const qq = proj(cc, Math.sin(b) * 1.3, 0, 9 + Math.cos(b) * 1.3); i ? kk.lineTo(qq.x, qq.y) : kk.moveTo(qq.x, qq.y); } kk.stroke(); kk.setLineDash([]); }
  });
  return u >= T0 && u < T1 ? `BULLET ×${SLOW.toFixed(1)}` : '×1.0';
});
