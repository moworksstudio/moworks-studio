// Hide-on-scroll header / 隠れるヘッダー
// Pack: lp2 (Landing page & web effects, part 2) · element 05 · Category: Scroll effects
// What happens: Scrolling down slides the header up out of view to free the screen; scrolling up even a little brings it straight back.
// Motion spec:
//   - A 511 × 168 viewport with a 34 px header fixed at the top. Scroll: 0→170 px over 0.3–1.1 s (down), 170→100 over 1.5–1.9 s (up), 100→230 over 2.4–2.9 s (down), 230→0 over 3.3–4.3 s (up), each with cubic ease-in-out. One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - If the last movement was downward the header hides; if upward it shows. Within 34 px of the top of the page it always shows.
//   - Hiding and showing both move it 36 px upward over 0.3 s (quadratic ease-in-out), like transform: translateY with a transition.
//   - While scrolled, a 12 px shadow sits under the header (opacity 0→0.14 over the first 30 px of scroll). The scrollbar thumb follows the scroll.
// When to use it: Use it when screen space is tight, as on phones, to give the content more room while keeping the menu one small scroll away.
// Tags: header, hide-on-scroll, headroom, navigation, sticky-header, scroll-direction
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#149
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 05
// 05 隠れるヘッダー（下へスクロールするとヘッダーが上へ引っ込み、上へ戻すとすっと出てくる）
CELLS[4] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const SY = t => steps([[0.3, 0], [1.1, 170], [1.5, 170], [1.9, 100], [2.4, 100], [2.9, 230], [3.3, 230], [4.3, 0]], t), y = SY(u);
  // 最後に動いた向きが「下」なら隠す（ページの一番上にいるときは出す）
  const hidden = t => { const yt = SY(t); if (yt < 34) return false; for (let i = 0, tt = t; i < 600; i++, tt -= 1 / 120) { const d = SY(tt) - SY(tt - 1 / 120); if (Math.abs(d) > 1e-4) return d > 0; } return false; };
  const q = trans(hidden, u, 0.3, eIOs);
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 48, 214, 110, 1, 8);
    txt(k, '旅の準備を、', 262, 84, `800 22px ${JP}`, C.ink); txt(k, 'まとめて。', 262, 114, `800 22px ${JP}`, C.ink); bars(k, 262, 128, 200, 2); btn(k, 262, 152, 100, 24, '詳しく見る');
    k.fillStyle = C.soft; k.fillRect(0, 196, VW, 260);
    txt(k, '人気のプラン', VW / 2, 232, `800 17px ${JP}`, C.ink, 'center');
    for (let r = 0; r < 2; r++) for (let i = 0; i < 3; i++) { const x = 34 + i * 152, y0 = 250 + r * 100; k.fillStyle = '#fff'; rr(k, x, y0, 136, 88, 8); k.fill(); pic(k, x + 8, y0 + 8, 120, 46, (i + r) % 3, 5); bars(k, x + 10, y0 + 64, 100, 2, C.faint, 5, 10); }
    k.restore();
    const hy = -36 * q;
    if (y > 0 && q < 1) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.14 * clamp(y / 30)})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 230, 0.42);
  });
};
