// Bird's-eye view / バーズアイ
// Pack: camera2 (Camera moves & transitions, part 2) · element 13 · Category: Angles & Framing
// What happens: Looking straight down from overhead: road, people, cars and trees appear as top-down shapes, and more comes into view as the camera rises.
// Motion spec:
//   - Camera: points straight down (pitch -π/2 = -90°) and rises from 9 m to 15 m, directly above the origin
//   - Timing: 0.5 s to 3.4 s (2.9 s), cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1))
//   - Ground: a 2 m grid and a 3.2 m wide road (white dashed centre line) drawn as one plane; six trees as circles and three buildings as squares, seen from above
//   - Subjects: a person (head circle and shoulder ellipse) walks 5.2 m up the right side of the road, a car (1.8×4 m) drives 6.8 m down the left side; both move in proportion to the progress p
//   - As the camera rises everything shrinks and more of the ground comes into view (10 High angle looks down at a slant)
//   - Return: hold 3.4–4.2 s, then cross-fade back to the start over the last 0.5 s; top-right readout shows "↓ 90° · 9m → 15m"
// When to use it: To show layout and paths like a diagram, or for a god's-eye overview.
// Tags: bird's-eye, birds eye, top-down, overhead, straight down, aerial, angle
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#181
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 13
// 13 バーズアイ: 真上（ピッチ -90°）から真下を見る。道・人・車・木がすべて上から見た形になる（10 ハイアングルは斜めから）
CELLS[12] = shot(camCell((k, w, h, p) => {
  const c = cam({ y: lerp(9, 15, p), pitch: -Math.PI / 2 });
  const ex = [[0, 0, 0, (kk, s) => {   // 地面（格子と道）を1枚のスプライトとして描く
    kk.fillStyle = C.g1; kk.fillRect(-1.6, -16, 3.2, 32);
    kk.strokeStyle = C.grid; kk.lineWidth = 1.3 / s; kk.beginPath();
    for (let i = -16; i <= 16; i += 2) { kk.moveTo(i, -16); kk.lineTo(i, 16); kk.moveTo(-16, i); kk.lineTo(16, i); }
    kk.stroke();
    kk.strokeStyle = '#fff'; kk.lineWidth = 2 / s; kk.setLineDash([0.6, 0.6]); kk.beginPath(); kk.moveTo(0, -16); kk.lineTo(0, 16); kk.stroke(); kk.setLineDash([]);
  }]];
  [[-4.5, -2.5, 1.0], [-5.5, 1.8, 1.2], [4.6, -0.6, 1.1], [5.2, 3.2, 0.9], [-4.2, 5.5, 1.0], [4.9, -4.4, 1.0]].forEach(([x, z, r]) => ex.push([x, 0, z, (kk) => { kk.fillStyle = C.g2; kk.beginPath(); kk.arc(0, 0, r, 0, 7); kk.fill(); kk.fillStyle = C.g3; kk.beginPath(); kk.arc(0, 0, 0.12, 0, 7); kk.fill(); }]));
  [[-8.5, -3, 3, 2.4], [8.6, 1, 3.4, 2.8], [-8.3, 2.5, 2.6, 2]].forEach(([x, z, bw, bd]) => ex.push([x, 0, z, (kk, s) => { kk.fillStyle = C.g1; kk.fillRect(-bw / 2, -bd / 2, bw, bd); kk.strokeStyle = C.g3; kk.lineWidth = 1.4 / s; kk.strokeRect(-bw / 2, -bd / 2, bw, bd); }]));
  ex.push([-0.8, 0, lerp(3.4, -3.4, p), (kk) => { kk.fillStyle = C.blue; rr(kk, -0.9, -2, 1.8, 4, 0.4); kk.fill(); kk.fillStyle = '#fff'; kk.fillRect(-0.7, 0.3, 1.4, 0.8); kk.fillRect(-0.7, -1.2, 1.4, 0.6); }]);
  ex.push([0.9, 0, lerp(-2.6, 2.6, p), (kk) => { kk.fillStyle = C.blue; kk.beginPath(); kk.ellipse(0, 0, 0.24, 0.12, 0, 0, 7); kk.fill(); kk.beginPath(); kk.arc(0, 0, 0.16, 0, 7); kk.fill(); }]);
  world(k, w, h, c, { grid: false, mountains: false, buildings: false, trees: false, sun: null, extra: ex });
}, p => `↓ 90° · ${lerp(9, 15, p).toFixed(0)}m`));
