// タイムラインアニメーション / Timeline on scroll
// パック: lp2 (LP・Webサイトの演出 第2弾) · element 01 · カテゴリ: スクロールで起きる演出
// 何が起きるか: ページを下へスクロールすると、年表の縦線が朱色で上から伸び、線の先が届いた年の点が塗られて、年と出来事が右から現れる。上へ戻すと逆にたどる。
// 動きの仕様:
//   - 表示域 511×168 のブラウザの窓。スクロール量: 0.3〜2.3秒で 0→190px、3.0〜4.3秒で 190→0（easeInOut 3乗）。4.7秒でひと回り（パックの周期9.4秒で2回）
//   - 年表の縦線は x=168。灰色の線（幅3）の上を朱色の線が伸びる。線の先＝スクロール量＋126px（いつも画面の上から126px、下から42pxの位置）
//   - 項目5つ（2018〜2026）。ページ上の y=72・126・180・234・288 に点（白・半径7.5・輪の線幅2.5）
//   - 線の先が点を過ぎてから22px伸びる間に（easeOut 3乗）: 輪が朱色になり、中の点が半径0→4に育ち、年（18px・朱）と出来事（15px）が右16pxから元の位置へ動きながら現れる
//   - 上へ戻すと線が縮み、同じ式で項目が消える。右端のスクロールバーも連動
// 使いどころ: 会社のあゆみ・沿革・導入までの流れなど、時間の順に並ぶ内容を読ませたいとき。
// タグ: timeline, scroll, history, milestones, scroll-driven, progress-line
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#145
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 01
// 01 タイムラインアニメーション（スクロールに合わせて年表の線が上から伸び、線が届いた年が順に現れる）
CELLS[0] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const y = steps([[0.3, 0], [2.3, 190], [3.0, 190], [4.3, 0]], u);
  const ROWS = [[72, '2018', '創業'], [126, '2020', '1号店オープン'], [180, '2022', '会員1万人'], [234, '2024', 'アプリ公開'], [288, '2026', '海外へ']];
  const LX = 168, L = y + 126;                         // 線の先（画面の下から 42px の所までページと一緒に伸びる）
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'HISTORY', 34, 34, `800 11px ${EN}`, C.acc, 'left', 'alphabetic', '2px');
    txt(k, 'あゆみ', 34, 60, `800 20px ${JP}`, C.ink);
    pic(k, 34, 76, 100, 62, 0, 6); bars(k, 34, 148, 90, 2);
    k.fillStyle = C.faint; rr(k, LX - 1.5, 52, 3, 248, 1.5); k.fill();
    k.fillStyle = C.acc; rr(k, LX - 1.5, 52, 3, clamp(L - 52, 0, 248), 1.5); k.fill();
    for (const [iy, yr, lab] of ROWS) {
      const q = eOut(clamp((L - iy) / 22));
      k.fillStyle = '#fff'; k.beginPath(); k.arc(LX, iy, 7.5, 0, 7); k.fill();
      k.lineWidth = 2.5; k.strokeStyle = q > 0 ? C.acc : C.faint; k.stroke();
      if (q > 0) { k.fillStyle = C.acc; k.beginPath(); k.arc(LX, iy, 4 * q, 0, 7); k.fill(); }
      k.save(); k.globalAlpha = q; k.translate(16 * (1 - q), 0);
      txt(k, yr, LX + 22, iy + 6, `800 18px ${EN}`, C.acc);
      txt(k, lab, LX + 82, iy + 5, `700 15px ${JP}`, C.ink);
      bars(k, LX + 82, iy + 15, 190, 1, C.faint, 5);
      k.restore();
    }
    k.restore();
    sbar(k, y / 190, 0.45);
  });
};
