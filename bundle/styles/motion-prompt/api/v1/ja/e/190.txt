// クロスカッティング / Cross-cutting
// パック: camera2 (カメラワークと切り替え 第2弾) · element 22 · カテゴリ: 切り替え・編集
// 何が起きるか: 同時に進む2つの場面（走る人物と走る車）を、だんだん短く交互に見せる。
// 動きの仕様:
//   - 2つの場面が同じ時刻を進む: A は走る人物（左0.3m、奥 15m → 4.5m、脚は u×12）、B は街を走る車（右1m、奥 17m → 4.5m）。どちらも4.7秒でカメラに近づく
//   - カット: 0 / 1.1 / 2.0 / 2.7 / 3.3 / 3.8 / 4.2 / 4.5秒で A・B を交互に直結。1.1 → 0.9 → 0.7 → 0.6 → 0.5 → 0.4 → 0.3秒と短くなり緊張を上げる
//   - 映っていない間もその場面の時間は進んでいる（戻ったとき位置が先に進んでいる）
//   - カメラ: Aは高さ1.3m・固定、手前に杭4本。Bは高さ1.6m・ピッチ+0.06rad・固定、街と太陽
//   - 画面下の帯: A（灰）B（青）が交互に並ぶ8区間と再生ヘッド。最後の0.3秒でAの開始状態へクロスフェード
//   - 右上に「SCENE A」「SCENE B」
// 使いどころ: 追跡や救出など、同時進行の緊張を高めるとき。
// タグ: cross-cutting, parallel editing, intercut, parallel action, chase, edit
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#190
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 22
// 22 クロスカッティング: 同時に進む2つの場面（走る人物・走る車）を、だんだん短く交互に見せる
const XC = [0, 1.1, 2.0, 2.7, 3.3, 3.8, 4.2, 4.5, CYC];
CELLS[21] = shot((k, w, h, u) => {
  const r = P(u, CYC - 0.3, CYC);
  const runA = (kk, t) => world(kk, w, h, cam({ y: 1.3 }), { person: { x: -0.3, z: lerp(15, 4.5, t / CYC), walk: t * 12 }, fg: [[-2.2, 3.5], [-2.2, 6.5], [-2.2, 9.5], [-2.2, 12.5]] });
  const carB = (kk, t) => world(kk, w, h, cam({ y: 1.6, pitch: 0.06 }), { city: true, mountains: false, trees: false, sun: [-26, 40, 150], extra: [[1.0, 0, lerp(17, 4.5, t / CYC), CAR[3]]] });
  let i = 0; while (u >= XC[i + 1]) i++;
  const isA = i % 2 === 0;
  blend(k, kk => (isA ? runA : carB)(kk, u), kk => runA(kk, 0), eIOs(r));   // 最後の0.3秒で A の開始状態へ戻す
  const segs = []; for (let j = 0; j < XC.length - 1; j++) segs.push([XC[j], XC[j + 1], j % 2 ? 'B' : 'A', j % 2 ? C.blue : C.g2]);
  strip(k, w, h, segs, u);
  return r > .5 || isA ? 'SCENE A' : 'SCENE B';
});
