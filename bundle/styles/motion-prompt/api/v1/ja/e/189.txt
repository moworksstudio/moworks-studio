// カットアウェイ / Cutaway
// パック: camera2 (カメラワークと切り替え 第2弾) · element 21 · カテゴリ: 切り替え・編集
// 何が起きるか: 話す人物の場面から、聞いている客席へ切り替えて戻る。差し込むのは別の対象。
// 動きの仕様:
//   - つなぎ方: 直結（0フレーム）が2回。1.7秒で別の対象へ、3.0秒で元の画へ戻る（1.3秒）
//   - 元の画（A）: 固定カメラ（高さ1.4m）。5.5m先で話す人物（腕と脚を 0.45·sin(u×6) rad 振る）、右0.9mに灰色の演台
//   - 別の対象（CUTAWAY）: 聞いている客席。5人（奥3.1〜4.6m、青と薄い青）が両腕を上げ、|sin(t×9+i)|×5cm で上下に弾む。背景なし（別の場所）
//   - 画面下の帯: A（灰）・別の対象（青）・A（灰）と再生ヘッド（17 インサートは同じ場面の細部）
//   - 右上に「SHOT A」→「CUTAWAY」→「SHOT A」。4.7秒で直結で最初に戻る
// 使いどころ: 反応を見せるとき、時間の省略や言い間違いのカットをごまかすとき。
// タグ: cutaway, reaction shot, b-roll, cut away, audience, edit
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#189
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 21
// 21 カットアウェイ: 話す人物から、別の対象（聞いている客席）へ切り替えて戻る。17 インサートは同じ場面の細部
const CUT = [1.7, 3.0];
CELLS[20] = shot((k, w, h, u) => {
  const away = u >= CUT[0] && u < CUT[1];
  if (!away) world(k, w, h, cam({ y: 1.4 }), { person: { x: 0, z: 5.5, walk: 0.45 * Math.sin(u * 6) }, extra: [[0.9, 0, 5.6, (kk) => { kk.fillStyle = C.g3; kk.fillRect(-0.3, 0, 0.6, 1.1); kk.fillRect(-0.4, 1.05, 0.8, 0.1); }]] });
  else {
    const t = u - CUT[0], ex = [];
    [[-1.6, 3.2], [-0.2, 3.6], [1.3, 3.1], [0.6, 4.4], [-1.1, 4.6]].forEach(([x, z], i) => ex.push([x, 0, z, (kk) => { kk.save(); kk.translate(0, Math.abs(Math.sin(t * 9 + i)) * 0.05); person(kk, 1, { armUp: true, col: i % 2 ? C.blue : C.blueM }); kk.restore(); }]));
    world(k, w, h, cam({ y: 1.6, pitch: -0.08 }), { buildings: false, trees: false, mountains: false, sun: null, extra: ex });
  }
  strip(k, w, h, [[0, CUT[0], 'A', C.g2], [CUT[0], CUT[1], '別の対象', C.blue], [CUT[1], CYC, 'A', C.g2]], u);
  return away ? 'CUTAWAY' : 'SHOT A';
});
